# ====== PiBoy DMG - Raspberry Pi 3B/3B+ display/audio block ==================
# Append this to the END of config.txt on the BATOCERA boot partition BEFORE the
# first boot. On the official bcm2837 image, also COMMENT OUT the existing
# "dtoverlay=vc4-kms-v3d" line: full KMS does not drive this DPI panel.

dtoverlay=vc4-fkms-v3d
max_framebuffers=2
extra_transpose_buffer=2
avoid_warnings=2
disable_overscan=1
disable_splash=1
boot_delay=0
initial_turbo=60
force_eeprom_read=0

# DPI pins: 22 GPIOs in ALT2
gpio=0-9,12-17,20-25=a2

# PiBoy PWM audio on GPIO 18/19
dtparam=audio=on
audio_pwm_mode=2
dtoverlay=audremap,pins_18_19

# 640x480 DPI panel. 70 Hz is what a real Pi 3 PiBoy runs reliably;
# the Pi 3 firmware does not hold the 85 Hz timing some guides give for the Pi 4.
hvs_set_dither=0x210
dpi_group=2
dpi_mode=87
dpi_output_format=0x070016
dpi_timings=640 1 80 80 80 480 1 13 13 13 0 0 0 70 0 32000000 1
#dpi_timings=640 1 130 130 130 480 1 13 13 13 0 0 0 60 0 32000000 1   # 60 Hz fallback
enable_dpi_lcd=1

# The Pi 3 enables composite out by default and reports it as always connected:
# Batocera 43 then treats the DPI panel as a secondary ("backglass") screen and
# you get a black screen with only the OSD. The PiBoy has no composite output.
enable_tvout=0

dtparam=act_led_trigger=none
dtparam=act_led_activelow=off
dtparam=pwr_led_trigger=none
dtparam=pwr_led_activelow=off

# An HDMI screen plugged in turns the internal panel off (one output at a time).
[edid=*]
enable_dpi_lcd=0
[all]
# ====== end of PiBoy DMG block ==============================================
